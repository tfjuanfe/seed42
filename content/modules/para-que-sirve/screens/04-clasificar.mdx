<CodePeek
  title="clasificador.py — el modelo completo"
  code={`
pesos = {"ganaste": 2.2, "premio": 1.8, "clave": 2.0,
         "profe": -1.5, "tarea": -1.2}   # aprendidos de miles de ejemplos

def es_estafa(mensaje):
    puntos = -1.6                # arranca escéptico
    for palabra in mensaje.lower().split():
        puntos += pesos.get(palabra, 0)
    return puntos > 0            # umbral de decisión
`}
